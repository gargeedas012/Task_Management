Q9. Find the sum IMDb rating for each genre
[
  {
    $unwind: {
      path: "$genres",
      includeArrayIndex: 'genresindex',
      preserveNullAndEmptyArrays: false
    }
  },
  {
    $group: {
      _id: "$genres",
      count: {
        "$sum": 1
      }
    }
  },
    {
    $project: {
      _id: 1,
      // genres: "$_id",
      count:1
    }
  }
]
var pipeline=new []
{
  new BsonDocument(
    "$unwind",{
      {"path","$genres"},
      { "includeArrayIndex", "genresindex" },
      { "preserveNullAndEmptyArrays", false }
    }
  ),
  new BsonDocument(
    "$group", {
      {"_id" , "$genres"},
      {"count", new BsonDocument ("$sum" ,1)}
    }
  ),
  new BsonDocument(
    "$project",
      new BsonDocument {
         { "_id", 1 },
         { "genres", "$_id },
         {"count" ,1}
      }
  )
}
var result= await movies.Aggregate<BsonDocument>(pipeline).ToListAsync();
//fluent api
var result = await movies.Aggregate()
    .Unwind("genres", new AggregateUnwindOptions<BsonDocument>
    {
        IncludeArrayIndex = "genresindex",
        PreserveNullAndEmptyArrays = false
    })
    .Group(
        new BsonDocument
        {
            { "_id", "$genres" },
            { "count", new BsonDocument("$sum", 1) }
        }
    )
    .Project(new BsonDocument
    {
        { "_id", 0 },
        { "genres", "$_id" },
        { "count", 1 }
    })
    .ToListAsync();

Q10. Find the highest-rated movie in each genre.
[
  {
    $unwind: {
      path: "$genres",
      preserveNullAndEmptyArrays: false
    }
  },
  {
    $group: {
      _id: "$genres",
      movie: {
        $first: "$title"
      },
     rating: {
        $first: "$imdb.rating"
      }
    }
  },
  {
    $sort: {
      rating: -1
    }
  },
    {
    $project: {
      _id: 0,
      genre: "$_id",
      movie: 1,
      rating: 1
    }
  }
]
//fluent api
var result = await movies.Aggregate()
    .Unwind("genres", new AggregateUnwindOptions<BsonDocument>
    {
        PreserveNullAndEmptyArrays = false
    })
    .Sort(new BsonDocument("imdb.rating", -1))
    .Group(new BsonDocument
    {
        { "_id", "$genres" },
        { "movie", new BsonDocument("$first", "$title") },
        { "rating", new BsonDocument("$first", "$imdb.rating") }
    })
    .Project(new BsonDocument
    {
        { "_id", 0 },
        { "genre", "$_id" },
        { "movie", 1 },
        { "rating", 1 }
    })
    .ToListAsync();
